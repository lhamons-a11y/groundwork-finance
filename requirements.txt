# Platform notes:
#   Windows: all packages below work as-is
#   macOS:   all packages below work as-is
#   Linux:   install system packages FIRST (see README), then pip install below

# Core web framework
flask>=3.1.0,<4.0
flask-cors>=4.0.0

# WSGI server. Flask's built-in server is for development and says so; this is
# what actually serves the app, in the bundle and in dev. Pure Python, so it
# freezes without a compiler, and cross-platform, unlike gunicorn.
waitress>=3.0.0

# Database
sqlalchemy>=2.0.0,<3.0

# Desktop window (cross-platform)
pywebview>=4.4.0

# HTTP client (for live price lookups + Anthropic AI)
requests>=2.31.0

# NOTE: there is deliberately no `anthropic` SDK dependency. The cloud AI path
# calls https://api.anthropic.com/v1/messages directly over HTTP and sets the
# anthropic-version header itself (server.py), so the SDK was never imported -
# it only added nine dependency trees (pydantic, httpx, anyio, ...) to every
# install and every CI run. Do not add it back without a call site.

# PDF statement extraction (Phase 4 — layout-to-text only, never interprets meaning)
pdfplumber>=0.11.0

# Memory check for the first-run AI choice (8 GB is the minimum for Groundwork AI, ENGINE_REPLACEMENT.md R-d)
psutil>=5.9.0

# The local AI engine (ENGINE_REPLACEMENT.md, item 31): runs the model inside Groundwork - no Ollama, no second
# program. Pinned to the version measured and packaged (0.3.35), from the project's prebuilt CPU wheels so no
# compiler is needed. If it is missing, Groundwork still runs and says AI is unavailable; a packaged build refuses
# to build without it (groundwork.spec).
--extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cpu
--only-binary llama-cpp-python
# Not on an Intel Mac: no tested wheel is published there, and Groundwork does not build an untested one
# (ENGINE_OTHER_PLATFORMS.md Q2). The application runs there; the engine reports local AI unavailable.
llama-cpp-python==0.3.35; sys_platform != "darwin" or platform_machine == "arm64"

# Item 113 (PANEL_REVIEW.md): voice commands, recognised on this computer by whisper.cpp (core/voice.py). Optional: without
# it, or without the model file (models/ggml-base.en.bin, scripts/fetch_whisper_model.py), voice reports unavailable.
pywhispercpp==1.5.1

# Groundwork's own voice (core/speak.py, Rob 2026-10-31): Kokoro, said on this computer through onnxruntime, with
# espeak-ng for pronunciation (pulled in by kokoro-onnx: onnxruntime, phonemizer, espeakng-loader). Optional: without
# it, or without the files (models/kokoro-v1.0.int8.onnx + voices-v1.0.bin, scripts/fetch_kokoro_model.py), read aloud
# uses the computer's own voices.
kokoro-onnx==0.6.1

# tkinter is NOT required on any platform. It used to be, for two startup
# error dialogs and a folder picker; those now use ctypes/osascript/zenity and
# pywebview respectively, and the PyInstaller specs exclude tkinter outright
# (it was ~10MB of tcl/tk per bundle). pywebview imports tkinter nowhere.
# The Downloads auto-import watcher (launch.py _start_downloads_watcher). Without it Groundwork runs and the watcher
# is off; every spec bundles its platform backend. Named here because it was missing: Windows builds had it only from
# the build machine, and the Linux build from this file shipped without it (ENGINE_OTHER_PLATFORMS.md section 7).
watchdog>=6.0.0,<7
# Item 57 (BANK_AGENT.md A4): bank logins live in the operating system's vault, never in Groundwork's files
keyring>=25.6.0,<26
# Item 81 (PERFECT_BACKUP.md): backups sealed with AES-256-GCM, keys stretched with scrypt. Already installed alongside
# the PDF reader; named here because the backups now depend on it directly.
cryptography>=42
